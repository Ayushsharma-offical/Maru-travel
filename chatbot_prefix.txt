  /* ===============================================================
   *  SECTION 3 — STATE
   *  A small in-memory + sessionStorage-backed state object so the
   *  conversation survives page navigation within the same tab.
   * =============================================================== */

  var STORAGE_KEY = "maruChatbotState_v1";

  var State = {
    isOpen: false,
    hasGreeted: false,
    history: [],          // [{ type: 'bot'|'user', text: '...' , time: 'HH:MM' }]
    currentCategoryId: null,
    unread: false
  };

  /**
   * Persist the current state to sessionStorage so a page reload or
   * navigation to another page on the same site keeps the chat
   * history intact for this browser tab.
   */
  function saveState() {
    try {
      window.sessionStorage.setItem(STORAGE_KEY, JSON.stringify(State));
    } catch (err) {
      // sessionStorage may be unavailable (private browsing, etc.) —
      // fail silently, the widget still works without persistence.
    }
  }

  /**
   * Restore state saved earlier in this browser tab's session, if any.
   */
  function loadState() {
    try {
      var raw = window.sessionStorage.getItem(STORAGE_KEY);
      if (raw) {
        var parsed = JSON.parse(raw);
        State.isOpen = false; // never auto-reopen on page load
        State.hasGreeted = parsed.hasGreeted || false;
        State.history = parsed.history || [];
        State.currentCategoryId = parsed.currentCategoryId || null;
        State.unread = false;
      }
    } catch (err) {
      // Ignore corrupted or missing storage.
    }
  }

  /* ===============================================================
   *  SECTION 4 — SMALL HELPERS
   * =============================================================== */

  /**
   * Returns the current time formatted as HH:MM (24-hour clock).
   */
  function nowTime() {
    var d = new Date();
    var h = d.getHours();
    var m = d.getMinutes();
    var hh = h < 10 ? "0" + h : "" + h;
    var mm = m < 10 ? "0" + m : "" + m;
    return hh + ":" + mm;
  }

  /**
   * Basic HTML-escaping so anything derived from user input (e.g. the
   * lead-capture form) can never break the DOM structure or inject
   * markup.
   */
  function escapeHtml(str) {
    return String(str)
      .replace(/&/g, "&amp;")
      .replace(/</g, "&lt;")
      .replace(/>/g, "&gt;")
      .replace(/"/g, "&quot;")
      .replace(/'/g, "&#039;");
  }

  /**
   * Very small helper to create a DOM element with attributes and
   * children in one call, used throughout the render functions below
   * to keep markup construction readable.
   */
  function el(tag, attrs, children) {
    var node = document.createElement(tag);
    attrs = attrs || {};
    for (var key in attrs) {
      if (!attrs.hasOwnProperty(key)) continue;
      if (key === "class") {
        node.className = attrs[key];
      } else if (key === "html") {
        node.innerHTML = attrs[key];
      } else if (key.indexOf("on") === 0 && typeof attrs[key] === "function") {
        node.addEventListener(key.slice(2).toLowerCase(), attrs[key]);
      } else {
        node.setAttribute(key, attrs[key]);
      }
    }
    children = children || [];
    for (var i = 0; i < children.length; i++) {
      if (children[i]) node.appendChild(children[i]);
    }
    return node;
  }

  /**
   * Finds a category object by its id.
   */
  function getCategory(id) {
    for (var i = 0; i < MaruConfig.categories.length; i++) {
      if (MaruConfig.categories[i].id === id) return MaruConfig.categories[i];
    }
    return null;
  }

  /**
   * Searches every question across every category for a case
   * insensitive substring match — powers the search box.
   */
  function searchQuestions(term) {
    var results = [];
    var needle = term.trim().toLowerCase();
    if (!needle) return results;
    MaruConfig.categories.forEach(function (cat) {
      cat.questions.forEach(function (item) {
        if (item.q.toLowerCase().indexOf(needle) !== -1 ||
            item.a.toLowerCase().indexOf(needle) !== -1) {
          results.push({ category: cat, item: item });
        }
      });
    });
    return results;
  }

  /**
   * Checks free-text user input against the small-talk table and
   * returns a matching reply string, or null if nothing matched.
   */
  function matchSmallTalk(text) {
    var lower = text.trim().toLowerCase();
    for (var i = 0; i < MaruConfig.smallTalk.length; i++) {
      var entry = MaruConfig.smallTalk[i];
      for (var j = 0; j < entry.triggers.length; j++) {
        if (lower.indexOf(entry.triggers[j]) !== -1) {
          return entry.reply;
        }
      }
    }
    return null;
  }

  /**
   * Appends a bot text bubble to the conversation, records it in
   * State.history (unless `record` is explicitly false), and scrolls
   * the body to the bottom.
   */
  function appendBotBubble(text, record) {
    var row = el("div", { class: "maru-chat-row bot" }, [
      el("div", { class: "maru-chat-bubble bot", html: escapeHtml(text) }),
      el("div", { class: "maru-chat-timestamp" }, [
        document.createTextNode(nowTime())
      ])
    ]);
    DOM.bodyEl.appendChild(row);
    scrollToBottom();
    if (record !== false) {
      State.history.push({ type: "bot", text: text, time: nowTime() });
      saveState();
    }
  }

  /**
   * Appends a user text bubble to the conversation (used when the
   * visitor clicks a question, types free text, or submits the lead
   * form).
   */
  function appendUserBubble(text, record) {
    var row = el("div", { class: "maru-chat-row user" }, [
      el("div", { class: "maru-chat-bubble user", html: escapeHtml(text) }),
      el("div", { class: "maru-chat-timestamp" }, [
        document.createTextNode(nowTime())
      ])
    ]);
    DOM.bodyEl.appendChild(row);
    scrollToBottom();
    if (record !== false) {
      State.history.push({ type: "user", text: text, time: nowTime() });
      saveState();
    }
  }

  /**
   * Shows an animated "typing…" bubble, then calls `callback` after a
   * short randomized delay and removes the indicator. This is what
   * makes bot replies feel conversational instead of instantaneous.
   */
  function showTypingThen(callback) {
    var dots = el("div", { class: "maru-typing-bubble" }, [
      el("span", { class: "maru-typing-dot" }),
      el("span", { class: "maru-typing-dot" }),
      el("span", { class: "maru-typing-dot" })
    ]);
    var row = el("div", { class: "maru-chat-row bot maru-typing-row" }, [dots]);
    DOM.bodyEl.appendChild(row);
    scrollToBottom();

    var delay = 450 + Math.floor(Math.random() * 400);
    window.setTimeout(function () {
      if (row.parentNode) row.parentNode.removeChild(row);
      callback();
    }, delay);
  }

  /**
   * Scrolls the message list to the latest message.
   */
  function scrollToBottom() {
    window.requestAnimationFrame(function () {
      DOM.bodyEl.scrollTop = DOM.bodyEl.scrollHeight;
    });
  }

  /**
   * Removes any currently-rendered option/chip/form block so a new
   * one can be rendered in its place. We tag the "live" interactive
   * block with data-maru-live="1" so it's easy to find and clear.
   */
  function clearLiveBlock() {
    var existing = DOM.bodyEl.querySelector('[data-maru-live="1"]');
    if (existing && existing.parentNode) {
      existing.parentNode.removeChild(existing);
    }
  }

  /**
   * Opens a category: shows an intro bot line, then lists its
   * questions, then a "back to main menu" chip.
   */
  function openCategory(categoryId) {
    var cat = getCategory(categoryId);
    if (!cat) return;

    State.currentCategoryId = categoryId;
    saveState();

    appendUserBubble(cat.label);
    showTypingThen(function () {
      appendBotBubble(cat.intro);
      renderQuestionList(cat);
    });
  }

  /**
   * Renders the list of questions within a category as option
   * buttons, plus a chip row to go back to the main menu.
   */
  function renderQuestionList(cat) {
    clearLiveBlock();
    var wrap = el("div", {
      class: "maru-chat-options",
      "data-maru-live": "1"
    });

    cat.questions.forEach(function (item) {
      var btn = el(
        "button",
        { class: "maru-option-btn", type: "button" },
        [
          el("span", {}, [document.createTextNode(item.q)]),
          el("i", { class: "fas fa-chevron-right trailing" })
        ]
      );
      btn.addEventListener("click", function (questionItem) {
        return function () {
          openAnswer(cat, questionItem);
        };
      }(item));
      wrap.appendChild(btn);
    });

    var chipRow = el("div", { class: "maru-chip-row" }, [
      backToMenuChip()
    ]);
    wrap.appendChild(chipRow);

    DOM.bodyEl.appendChild(wrap);
    scrollToBottom();
  }

  /**
   * Displays the answer to a single question, then offers
   * "was this helpful" follow-up chips.
   */
  function openAnswer(cat, item) {
    appendUserBubble(item.q);
    showTypingThen(function () {
      appendBotBubble(item.a);
      renderPostAnswerChips(cat);
    });
  }

  /**
   * After every answer we show three quick chips: more questions in
   * the same category, back to the main menu, or talk to a human on
   * WhatsApp. This keeps the conversation moving without forcing the
   * visitor to type anything.
   */
  function renderPostAnswerChips(cat) {
    clearLiveBlock();
    var wrap = el("div", {
      class: "maru-chip-row",
      "data-maru-live": "1"
    });

    var moreBtn = el("button", { class: "maru-chip-btn primary", type: "button" }, [
      document.createTextNode("More on " + cat.label)
    ]);
    moreBtn.addEventListener("click", function () {
      appendUserBubble("More on " + cat.label);
      showTypingThen(function () {
        renderQuestionList(cat);
      });
    });

    var whatsappBtn = el("button", { class: "maru-chip-btn", type: "button" }, [
      document.createTextNode("Talk to a human")
    ]);
    whatsappBtn.addEventListener("click", function () {
      window.open("https://wa.me/" + MaruConfig.contact.whatsappNumber, "_blank");
    });

    wrap.appendChild(moreBtn);
    wrap.appendChild(backToMenuChip());
    wrap.appendChild(whatsappBtn);

    DOM.bodyEl.appendChild(wrap);
    scrollToBottom();
  }

  /**
   * Builds a reusable "Back to main menu" chip button.
   */
  function backToMenuChip() {
    var chip = el("button", { class: "maru-chip-btn", type: "button" }, [
      document.createTextNode("\u2190 Main menu")
    ]);
    chip.addEventListener("click", function () {
      appendUserBubble("Back to main menu");
      State.currentCategoryId = null;
      saveState();
      showTypingThen(function () {
        appendBotBubble("Sure — here's the main menu again. What would you like to know?");
        renderMainMenu();
      });
    });
    return chip;
  }
